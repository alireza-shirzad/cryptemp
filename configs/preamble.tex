% Build configuration flags
%   full   : full version. Uses a single-column article layout (11pt, letter,
%            wide margins, with a table of contents) and shows the real author
%            names. This is the eprint / arXiv "full version".
%   lncs   : submission uses the LNCS (llncs) style.
%   usenix : submission uses the USENIX style.
%   sp     : submission uses the IEEE S&P (IEEEtran conference/compsoc) style.
%            The lncs/usenix/sp flags only take effect when \fullfalse; the
%            submission is then anonymized.
%   notes  : show inline author comments (\ali, \pratyush, \sriram, ...).
%            When false, these comments are hidden.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\newif\iffull
\newif\iflncs
\newif\ifusenix
\newif\ifsp
\newif\ifnotes

% ---- Select the configuration here ----
\lncstrue
% \usenixtrue
% \sptrue
\fulltrue
\notestrue
% ---------------------------------------

\synctex=1

% The full version is a single-column article, independent of the venue style.
% Otherwise, the (anonymized) submission uses the selected venue style.
\iffull
  \documentclass[11pt,letterpaper]{article}
\else
  \iflncs
    \documentclass[letter]{llncs}
    \pagestyle{plain}
  \fi
  \ifusenix
    \documentclass[letterpaper,twocolumn,10pt]{article}
    \usepackage{styles/usenix}
  \fi
  \ifsp
    % IEEE S&P (Oakland): IEEEtran conference/compsoc, US letter.
    \documentclass[conference,compsoc,letterpaper]{IEEEtran}
  \fi
\fi

\usepackage{multicol}
